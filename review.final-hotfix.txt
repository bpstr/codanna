import base64, gzip, hashlib, json, os
from pathlib import Path
encoded = Path(os.environ['RUNNER_TEMP'], 'review.final-admission.b64').read_bytes().replace(b'NVPKL+Lk', b'NVPKLk')
raw = gzip.decompress(base64.b64decode(encoded, validate=True))
assert hashlib.sha256(raw).hexdigest() == '578ec9ff426ab74ed935426db4a74388c6c7b2523e16d3c7675227da25cccb2a'
for change in json.loads(raw):
    path = Path(change['path'])
    assert str(path) == 'src/mcp/auth.rs'
    old = path.read_bytes()
    assert hashlib.sha256(old).hexdigest() == change['old']
    lines = old.decode().splitlines(keepends=True)
    for first, last, replacement in reversed(change['edits']):
        lines[first:last] = replacement.splitlines(keepends=True)
    new = ''.join(lines).encode()
    assert hashlib.sha256(new).hexdigest() == change['new']
    path.write_bytes(new)
